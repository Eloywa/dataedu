Средний балл по базам данных в каждой группе

SELECT g.name, round(avg(m.mark), 2) AS avg_mark
FROM marks m
JOIN students s ON s.id = m.student_id
JOIN groups g ON g.id = s.group_id
WHERE m.subject = 'Базы данных'
GROUP BY g.name
ORDER BY avg_mark DESC;
